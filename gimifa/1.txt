<script>
    // 綁定 DOM 元素
    const imageInput = document.getElementById('imageInput');
    const fillAllInput = document.getElementById('fillAllInput');
    let currentTarget = null; // 記錄當前點擊的是哪一個圖片容器

    // 單張上傳觸發邏輯
    function triggerUpload(el) {
        currentTarget = el.querySelector('.preview'); // 指向該容器內的 img 標籤
        imageInput.value = ''; // 重置 value 確保選同一張圖也會觸發變更
        imageInput.click(); // 模擬點擊
    }

    // 單張上傳處理程序
    imageInput.addEventListener('change', function(e) {
        handleFileSelect(e.target.files[0], (dataUrl) => {
            if (currentTarget) updatePhotoBox(currentTarget, dataUrl);
        });
    });

    // 一鍵填滿觸發邏輯
    function triggerFillAll() {
        fillAllInput.value = ''; 
        fillAllInput.click();
    }

    // 一鍵填滿處理程序
    fillAllInput.addEventListener('change', function(e) {
        handleFileSelect(e.target.files[0], (dataUrl) => {
            const allPreviews = document.querySelectorAll('.preview');
            allPreviews.forEach(img => updatePhotoBox(img, dataUrl)); // 迴圈填入所有圖片
        });
    });

    // 更新圖片到格子的核心函數
    function updatePhotoBox(imgEl, dataUrl) {
        imgEl.src = dataUrl; // 設定圖片來源
        imgEl.style.display = 'block'; // 顯示圖片
        const parent = imgEl.parentElement;
        parent.style.background = 'transparent'; // 移除灰背景
        parent.style.border = 'none'; // 隱藏邊框以求轉圖美觀
    }

    // 檔案讀取器封裝
    function handleFileSelect(file, callback) {
        if (file) {
            const reader = new FileReader();
            // 讀取成功後回傳 Base64 資料流
            reader.onload = (ev) => callback(ev.target.result);
            reader.readAsDataURL(file);
        }
    }

    // 核心轉圖函數
    async function processLayout() {
        const a4 = document.getElementById('a4-page');
        const saveBtn = document.getElementById('saveBtn');
        const isMobile = /iPhone|iPad|iPod|Android/i.test(navigator.userAgent);
        
        saveBtn.innerText = "正在生成超高畫質圖...";
        saveBtn.disabled = true;

        // 稍微延遲，確保圖片已被瀏覽器完全解碼
        await new Promise(r => setTimeout(r, 500));

        // 開始進行 HTML 轉 Canvas
        html2canvas(a4, { 
            scale: 5, // 強制 5 倍解析度輸出（確保 A4 列印不糊）
            useCORS: true, // 開啟跨域支持
            logging: false, // 關閉排錯日誌
            backgroundColor: '#BC4C41', // 設定背景色
            windowWidth: 794, // 固定模擬 A4 寬度像素
            windowHeight: 1123, // 固定模擬 A4 高度像素
            onclone: (clonedDoc) => {
                // 在克隆出的節點上移除預覽用的 transform，否則轉出來的圖會只有一小塊
                clonedDoc.getElementById('a4-page').style.transform = 'none';
            }
        }).then(canvas => {
            // 將畫布轉為 PNG 格式，不進行任何畫質壓縮
            const dataUrl = canvas.toDataURL('image/png');
            
            if (isMobile) {
                // 手機版：顯示黑色遮罩讓使用者手動存圖
                document.getElementById('final-image').src = dataUrl;
                document.getElementById('result-overlay').style.display = 'flex';
            } else {
                // 電腦版：自動下載檔案
                const link = document.createElement('a');
                link.download = `春聯排版_高清_${new Date().getTime()}.png`;
                link.href = dataUrl;
                link.click();
            }
            
            saveBtn.innerText = "💾 下載 A4 圖片";
            saveBtn.disabled = false;
        }).catch(err => {
            console.error(err);
            alert("生成失敗，可能是圖片過大，請嘗試換一張圖。");
            saveBtn.disabled = false;
        });
    }

    // 清除畫面上所有圖片的邏輯
    function clearAllPhotos() {
        if (confirm('確定要清除所有照片嗎？')) {
            document.querySelectorAll('.preview').forEach(img => {
                img.src = '';
                img.style.display = 'none';
                const parent = img.parentElement;
                parent.style.background = 'rgba(255,255,255,0.1)';
                parent.style.border = '0.2mm solid rgba(255,255,255,0.3)';
            });
            imageInput.value = '';
            fillAllInput.value = '';
        }
    }
</script>
