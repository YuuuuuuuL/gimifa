<script>
    const a4 = document.getElementById('a4-page');
    const fInput = document.getElementById('fInput');
    const fillAllInput = document.getElementById('fillAllInput');
    let targetBox = null;

    function makeBox(cls) {
        const div = document.createElement('div');
        div.className = 'chun-lian ' + cls;
        div.onclick = (e) => { e.stopPropagation(); targetBox = div; fInput.value = ''; fInput.click(); };
        return div;
    }

    function addBestCombo15() {
        const row = document.createElement('div');
        row.className = 'combo-row';
        row.appendChild(makeBox('s15'));
        const v = document.createElement('div');
        v.className = 'v-stack';
        v.appendChild(makeBox('s5')); v.appendChild(makeBox('s5')); v.appendChild(makeBox('s5'));
        row.appendChild(v);
        canFit(row);
    }

    function addBestCombo10() {
        const row = document.createElement('div');
        row.className = 'combo-row';
        row.appendChild(makeBox('s10'));
        for(let i=0; i<2; i++) {
            const v = document.createElement('div');
            v.className = 'v-stack';
            v.appendChild(makeBox('s5')); v.appendChild(makeBox('s5'));
            row.appendChild(v);
        }
        canFit(row);
    }

    function addSingle(cls) { canFit(makeBox(cls)); }

    function canFit(el) {
        a4.appendChild(el);
        if (a4.scrollHeight > a4.clientHeight + 2) {
            alert("空間不足！"); a4.removeChild(el); return false;
        }
        return true;
    }

    function triggerFillAll() {
        if (a4.children.length === 0) return alert("請先新增春聯格子！");
        fillAllInput.click();
    }

    fillAllInput.onchange = (e) => handleFiles(e, true);
    fInput.onchange = (e) => handleFiles(e, false);

    function handleFiles(e, isAll) {
        const file = e.target.files[0];
        if (!file) return;
        const reader = new FileReader();
        reader.onload = (event) => processImage(event.target.result, isAll);
        reader.readAsDataURL(file);
    }

    function processImage(dataUrl, isAll) {
        const img = new Image();
        img.onload = function() {
            const canvas = document.createElement('canvas');
            const ctx = canvas.getContext('2d');
            
            // 提高原始圖片品質上限
            const maxDim = 3500; 
            let w = img.width, h = img.height;
            if (w > maxDim || h > maxDim) {
                const ratio = Math.min(maxDim / w, maxDim / h);
                w *= ratio; h *= ratio;
            }
            
            canvas.width = w; canvas.height = h;
            ctx.drawImage(img, 0, 0, w, h);
            
            // 使用無損 PNG 確保畫質
            const finalDataUrl = canvas.toDataURL('image/png');

            if (isAll) {
                document.querySelectorAll('.chun-lian').forEach(box => updateBox(box, finalDataUrl));
            } else if (targetBox) {
                updateBox(targetBox, finalDataUrl);
            }
        };
        img.src = dataUrl;
    }

    function updateBox(box, src) {
        box.innerHTML = `<img src="${src}">`;
        box.style.border = "none";
        box.style.background = "transparent";
    }

    function handleExport() {
        if (a4.children.length === 0) return alert("請先新增內容");
        
        const btn = document.getElementById('exportBtn');
        btn.innerText = "⚡ 高清處理中 (請勿關閉)...";
        btn.disabled = true;

        // scale: 4 提供約 300-400 DPI 的輸出畫質
        html2canvas(a4, { 
            scale: 4, 
            useCORS: true,
            backgroundColor: '#BC4C41'
        }).then(canvas => {
            const dataUrl = canvas.toDataURL('image/png');
            const isMobile = /iPhone|iPad|iPod|Android/i.test(navigator.userAgent);
            
            if (isMobile) {
                document.getElementById('final-image').src = dataUrl;
                document.getElementById('result-overlay').style.display = 'flex';
            } else {
                const link = document.createElement('a');
                link.download = '春聯高品質列印_' + Date.now() + '.png';
                link.href = dataUrl;
                link.click();
            }
            btn.innerText = "📥 下載高畫質 A4 列印圖";
            btn.disabled = false;
        });
    }
</script>
